DAY 08 - Variables, VPC, Probes & Loops

--- Ansible ---
- name: Install Pathnex packages
  hosts: all
  become: yes
  vars:
    packages:
      - tree
      - unzip
      - vim
  tasks:
    - name: Install packages
      yum:
        name: "{{ packages }}"
        state: present


--- Terraform ---
provider "aws" {
  region = "us-east-1"
}

resource "aws_vpc" "pathnex_vpc" {
  cidr_block = "10.10.0.0/16"

  tags = {
    Name = "Pathnex-VPC"
  }
}

resource "aws_subnet" "pathnex_subnet" {
  vpc_id                  = aws_vpc.pathnex_vpc.id
  cidr_block              = "10.10.1.0/24"
  map_public_ip_on_launch = true
}


--- Kubernetes ---
apiVersion: v1
kind: Pod

metadata:
  name: pathnex-probe-pod

spec:
  containers:
    - name: web
      image: nginx

      livenessProbe:
        httpGet:
          path: /
          port: 80

      readinessProbe:
        httpGet:
          path: /

--- Jenkins ---
parameters {
    string(
        name: 'ENVIRONMENT',
        defaultValue: 'dev'
    )
}

stage('Build') {
    when {
        branch 'main'
    }
    steps {
        sh 'mvn clean compile'
    }
}

stage('Deploy') {
    when {
        expression {
            params.ENVIRONMENT == 'dev'
        }
    }
    steps {
        sh 'echo Deploying to dev'
    }
}


--- GitLab CI ---
stages:
  - build
  - deploy

build:
  stage: build
  script:
    - mvn clean compile
  only:
    - main

deploy:
  stage: deploy
  script:
    - echo "Deploying to dev"
  only:
    variables:
      - $CI_ENVIRONMENT_NAME == "dev"
